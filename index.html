alter table groups add column if not exists owner uuid; -- 2) ユーザー検索用プロフィール create table if not exists profiles ( user_id uuid primary key, name text not null, updated_at timestamptz default now() ); alter table profiles enable row level security; drop policy if exists "profiles read" on profiles; drop policy if exists "profiles write own" on profiles; drop policy if exists "profiles update own" on profiles; create policy "profiles read" on profiles for select to authenticated using (true); create policy "profiles write own" on profiles for insert to authenticated with check (user_id = auth.uid()); create policy "profiles update own" on profiles for update to authenticated using (user_id = auth.uid()); -- 3) 招待 create table if not exists invites ( id uuid primary key default gen_random_uuid(), group_id text not null, group_name text, to_user uuid not null, from_name text, status text default 'pending', created_at timestamptz default now() ); alter table invites enable row level security; drop policy if exists "invites insert" on invites; drop policy if exists "invites read own" on invites; drop policy if exists "invites update own" on invites; create policy "invites insert" on invites for insert to authenticated with check (true); create policy "invites read own" on invites for select to authenticated using (to_user = auth.uid()); create policy "invites update own" on invites for update to authenticated using (to_user = auth.uid()); -- 4) 楽譜・動画の保存先 (公開バケット) insert into storage.buckets (id, name, public) values ('scores','scores',true) on conflict (id) do nothing; drop policy if exists "scores upload" on storage.objects; drop policy if exists "scores read" on storage.objects; create policy "scores upload" on storage.objects for insert to authenticated with check (bucket_id = 'scores'); create policy "scores read" on storage.objects for select using (bucket_id = 'scores');
